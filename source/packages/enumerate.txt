function enumerate(itx, start)
    define x []
    define p start
    define i 0

    loop
        pushlocal i
        pushlocal itx
        calllocalnative iterlen

        if i >= poplocal
            exit 3
        end 

        pushlocal x
        pushlocal p
        calllocalnative iterappend
        poplocal x

        pushlocal itx
        pushlocal i
        calllocalnative getelement

        pushlocal x
        calllocalnative swap

        calllocalnative iterappend

        calllocalnative totuple
        poplocal x
        push x
        set x []

        pushlocal p 
        calllocalnative inc 
        pushlocal i
        calllocalnative inc
        poplocal i
        poplocal p 
    end
end
