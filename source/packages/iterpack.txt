function iterpack(depth, type)
    if depth < 0
        pushlocal "Cannot pack a negative amount of items into an iterable"
        calllocalnative print
        exit 2
    end

    define x []
    pushlocal depth
    loop
        pushlocal x
        pushlocal pop
        calllocalnative iterappend
        poplocal x

        calllocalnative dec
        calllocalnative dup

        if poplocal == 0
            if type == "T"
                pushlocal x
                calllocalnative totuple
                poplocal x
                push x
            elif type == "L"
                push x
            end
            exit 3
        end
    end
end

function iterpackall(type)
    define x 0
    callnative depth
    pop x
    call iterpack x type
end
