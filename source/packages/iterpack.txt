function iterpack(depth)
    if depth < 0
        pushlocal "Cannot pack a negative amount of items into an iterable"
        calllocalnative print
        exit 2
    end
    
    define x []
    pushlocal depth
    loop
        pushlocal x
        pushlocal pop
        calllocalnative iterappend
        poplocal x

        calllocalnative dec
        calllocalnative dup

        if poplocal == 0
            push x
            exit 3
        end
    end
end
