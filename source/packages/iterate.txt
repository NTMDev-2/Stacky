function iterate(itx, func)
    define i 0

    loop
        pushlocal itx
        calllocalnative iterlen
        if i >= poplocal
            exit 3
        end
        
        pushlocal itx
        pushlocal i
        calllocalnative getelement
        
        push poplocal

        call func
        
        pushlocal i
        calllocalnative inc
        poplocal i

    end
end
